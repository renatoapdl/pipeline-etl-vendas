SELECT
  id,
  nome,
  CASE 
    WHEN idade < 0 THEN NULL 
    ELSE idade 
  END AS idade,
  CASE 
    WHEN email NOT LIKE '%@%' THEN NULL 
    ELSE email 
  END AS email,
  CASE 
    WHEN LENGTH(telefone) != 10 THEN NULL
    ELSE CONCAT('(', SUBSTRING(telefone, 1, 2), ') ', SUBSTRING(telefone, 3, 4), '-', SUBSTRING(telefone, 7, 4))
  END AS telefone,
  endereco,
  cidade,
  TRIM(estado) AS estado,
  CASE
    WHEN TRIM(estado) = 'AM' THEN 'Amazonas'
    WHEN TRIM(estado) = 'BA' THEN 'Bahia'
    WHEN TRIM(estado) = 'CE' THEN 'Ceará'
    WHEN TRIM(estado) = 'ES' THEN 'Espirito Santo'
    WHEN TRIM(estado) = 'MG' THEN 'Minas Gerais'
    WHEN TRIM(estado) = 'PR' THEN 'Paraná'
    WHEN TRIM(estado) = 'PE' THEN 'Pernambuco'
    WHEN TRIM(estado) = 'RJ' THEN 'Rio de Janeiro'
    WHEN TRIM(estado) = 'RS' THEN 'Rio Grande do Sul'
    WHEN TRIM(estado) = 'SP' THEN 'São Paulo'
    ELSE NULL
  END AS estado_completo,
  CASE
    WHEN REGEXP_LIKE(cep, '[0-9]{5}-[0-9]{3}') THEN cep
    ELSE '00000-000'
  END AS cep,
  CASE
    WHEN LOWER(pais) LIKE 'br%l'  THEN 'Brasil'
    ELSE NULL
  END AS pais,
  empresa,
  cargo,
  CASE
    WHEN salario < 0 THEN NULL
    ELSE salario
  END AS salario,
  CASE
    WHEN TO_DATE(data_contratacao, 'yyyy-MM-dd') IS NOT NULL THEN TO_DATE(data_contratacao, 'yyyy-MM-dd')
    ELSE '1900-00-00'
  END AS data_contratacao,
  CASE
    WHEN TO_DATE(data_nascimento, 'yyyy-MM-dd') IS NOT NULL THEN TO_DATE(data_nascimento) 
    ELSE '1900-00-00'
  END as data_nascimento,
  CASE
    WHEN TO_DATE(data_nascimento) IS NULL THEN NULL
    ELSE FLOOR(DATEDIFF(DAY, data_nascimento, (CURRENT_DATE())) / 365.23)
  end as idade_real,
  CASE
    WHEN REGEXP_LIKE(cpf, '[0-9]{11}') THEN
      CONCAT(SUBSTRING(cpf, 1, 3), '.', SUBSTRING(cpf, 4, 3), '.', SUBSTRING(cpf, 7, 3), '-', SUBSTRING(cpf, 10, 2))
    ELSE '000.000.000-00'
  END AS cpf,
  cartao_credito,
  CASE
    WHEN pontuacao_credito BETWEEN 300 AND 850 THEN pontuacao_credito
    ELSE NULL
  END AS pontuacao_credito,
  CASE 
    WHEN divida < 0 THEN NULL 
    ELSE divida 
  END AS divida,
  CASE
    WHEN UPPER(genero) IN ('M', 'F') THEN genero
    ELSE NULL
  END AS genero
FROM dados_brutos;
